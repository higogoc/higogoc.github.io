{%- comment -%} Renders a row of badges from a list of keys. Usage: {% include badges.html list=item.badges %} Keys resolve through _data/badges.yml. "award:1st Prize" uses the text after the colon as the label and the key before it for the styling tier. {%- endcomment -%} {%- if include.list and include.list.size > 0 -%} {%- for key in include.list -%} {%- assign parts = key | split: ":" -%} {%- assign bkey = parts[0] -%} {%- assign def = site.data.badges[bkey] -%} {%- if parts.size > 1 -%} {%- assign label = key | remove_first: bkey | remove_first: ":" -%} {%- else -%} {%- assign label = def.label | default: key -%} {%- endif -%} {%- assign tier = def.tier | default: "neutral" -%} {{ label }} {%- endfor -%} {%- endif -%}
